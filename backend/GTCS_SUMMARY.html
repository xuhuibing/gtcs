快速启动

# 开发环境
cd backend
python3 -m uvicorn app.main:app --host 0.0.0.0 --port 8008

# 前端 (另一个终端)
cd frontend
npm run dev

# 生产部署
docker-compose up -d

# 数据导入
python3 scripts/import_tariff_data.py     # 全部
python3 scripts/import_tariff_data.py --country US  # 仅美国
python3 scripts/import_tariff_data.py --clean       # 清理重导
python3 scripts/seed_data.py             # 初始化种子数据

# 访问
# API:    http://localhost:8008/docs
# 前端:   http://localhost:5173 (开发)
#         http://localhost:8008 (生产)
# 默认:   admin / admin123
